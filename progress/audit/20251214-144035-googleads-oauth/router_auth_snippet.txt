function normalizeEmail(email) {
  return String(email || '').trim().toLowerCase();
}


// Frontend contract (CoolBits chat):
// - Clicking "Sign in with Google" opens /api/auth/google/start in a popup.
// - The callback page posts { token, user } via window.postMessage to the opener.
// - chat.js on the frontend listens for source === 'coolbits-oauth', applies auth state, and refreshes usage.

function parseCookies(header) {
  const list = {};
  if (!header) return list;
  header.split(';').forEach((cookie) => {
    const parts = cookie.split('=');
    const key = parts.shift()?.trim();
    if (!key) return;
    const value = decodeURIComponent(parts.join('='));
    list[key] = value;
  });
  return list;
}

function base64UrlEncode(obj) {
  return Buffer.from(JSON.stringify(obj)).toString('base64url');
}

function base64UrlDecode(str) {
  try {
    return JSON.parse(Buffer.from(str, 'base64url').toString('utf8'));
  } catch (e) {
    return null;
  }
}

function sendOAuthError(res, reason) {
  const html = `<!doctype html><html><body><script>(function(){var msg={source:'coolbits-oauth',error:'${reason}'};if(window.opener&&window.opener.postMessage){window.opener.postMessage(msg,'*');}window.close();})();</script></body></html>`;
  res.set('Content-Type', 'text/html').status(200).send(html);
}

async function handleGoogleStart(req, res) {
  if (!GOOGLE_CLIENT_ID || !GOOGLE_CLIENT_SECRET) {
    return res.status(503).json({ error: 'google_oauth_not_configured' });
  }
  const returnTo = typeof req.query.returnTo === 'string' ? req.query.returnTo : '/chat';
  const stateToken = crypto.randomBytes(16).toString('hex');
  const statePayload = base64UrlEncode({ s: stateToken, returnTo });
  const authUrl = oauthClient.generateAuthUrl({
    access_type: 'offline',
    prompt: 'consent',
    scope: ['openid', 'email', 'profile'],
    redirect_uri: GOOGLE_REDIRECT_URI,
    state: statePayload,
  });
  const isSecure = req.protocol === 'https' || req.headers['x-forwarded-proto'] === 'https';
  res.cookie('cb_oauth_state', stateToken, {
    httpOnly: true,
    sameSite: 'lax',
    secure: isSecure,
    maxAge: 10 * 60 * 1000,
  });
  const html = `<!doctype html><html><body><script>location.replace('${authUrl}');</script></body></html>`;
  res.set('Cache-Control', 'no-store');
  return res.status(200).send(html);
}

router.get('/google/start', (req, res) => {
  handleGoogleStart(req, res).catch((err) => {
    console.error('[AUTH] google/start error', err);
    res.status(500).json({ error: 'google_oauth_start_failed' });
  });
});

router.post('/google/start', (req, res) => {
  handleGoogleStart(req, res).catch((err) => {
    console.error('[AUTH] google/start POST error', err);
    res.status(500).json({ error: 'google_oauth_start_failed' });
  });
});

router.get('/google/callback', async (req, res) => {
  try {
    const { code, state, error: googleError } = req.query;
    if (googleError) {
      return sendOAuthError(res, String(googleError));
    }
    if (!code || !state) {
      return sendOAuthError(res, 'missing_code_or_state');
    }
    const stateData = base64UrlDecode(String(state));
    if (!stateData?.s) {
      return sendOAuthError(res, 'invalid_state');
    }
    const cookies = parseCookies(req.headers.cookie || '');
    const cookieState = cookies['cb_oauth_state'];
    res.clearCookie('cb_oauth_state');
    if (!cookieState || cookieState !== stateData.s) {
      return sendOAuthError(res, 'invalid_state');
    }

    const { tokens } = await oauthClient.getToken({ code, redirect_uri: GOOGLE_REDIRECT_URI });
    if (!tokens?.id_token) {
      return sendOAuthError(res, 'missing_id_token');
    }
    const ticket = await oauthClient.verifyIdToken({
      idToken: tokens.id_token,
      audience: GOOGLE_CLIENT_ID,
    });
    const payload = ticket.getPayload() || {};
    const email = (payload.email || '').toLowerCase();
    if (!email) {
      return sendOAuthError(res, 'email_required');
    }
    const emailVerified = Boolean(payload.email_verified);
    const now = new Date().toISOString();
    const existing = await getUserByEmail(email);
    const user = await upsertUser({
      email,
      emailVerified,
      emailVerifiedAt:
        emailVerified
          ? (existing?.emailVerifiedAt || now)
          : (existing?.emailVerifiedAt || null),
    });
    const safeUser = {
      id: user.id || user.email,
      email: user.email,
      planId: user.planId,
      tokensRemaining: user.tokensRemaining,
      totalUsed: user.totalUsed ?? 0,
      emailVerified: Boolean(user.emailVerified),
      emailVerifiedAt: user.emailVerifiedAt || null,
    };
    const token = signUser({ email: user.email, plan: user.planId });
    const returnTo =
      typeof stateData.returnTo === 'string' && stateData.returnTo.startsWith('/')
        ? stateData.returnTo
        : '/chat';
    const payloadToSend = { token, user: safeUser, returnTo };
    // Also set a cookie so /auth/me works without explicit Authorization header.
    const isSecure = req.protocol === 'https' || req.headers['x-forwarded-proto'] === 'https';
    const cookieDomain = process.env.AUTH_COOKIE_DOMAIN || null;
    res.cookie('cb_token', token, {
      httpOnly: true,
      secure: isSecure,
      sameSite: 'lax',
      path: '/',
      ...(cookieDomain ? { domain: cookieDomain } : {}),
      maxAge: 30 * 24 * 60 * 60 * 1000,
    });
    const payloadJson = JSON.stringify(payloadToSend)
      .replace(/</g, '\u003c')
      .replace(/>/g, '\u003e')
      .replace(/&/g, '\u0026');
    const html = `<!doctype html><html><body><script>(function(){var payload=${payloadJson};try{if(window.opener&&!window.opener.closed){window.opener.postMessage({source:'coolbits-oauth',payload:payload},'*');}}catch(err){console.error('[OAUTH CALLBACK] postMessage error',err);}window.close();})();</script></body></html>`;
    res.set('Content-Type', 'text/html').send(html);
  } catch (error) {
    logError(error);
    return sendOAuthError(res, 'google_oauth_failed');
  }
});

router.post('/mock-login', async (req, res) => {
  try {
    const email = normalizeEmail(req.body?.email);
    if (!email || !email.includes('@')) {
      return res.status(400).json({ error: 'Invalid email' });
    }
    const existing = await getUserByEmail(email);
    const user = await upsertUser({
      email,
      planId: existing?.planId || PLANS.STARTER_FREE.id,
      tokensRemaining:
        typeof existing?.tokensRemaining === 'number'
          ? existing.tokensRemaining
          : PLANS.STARTER_FREE.tokensIncluded,
    });
    const token = signUser({ email: user.email, plan: user.planId });
    return res.json({
      token,
      user: {
        email: user.email,
        plan: user.planId,
        tokensRemaining: user.tokensRemaining,
        totalUsed: user.totalUsed,
        emailVerified: Boolean(user.emailVerified),
        emailVerifiedAt: user.emailVerifiedAt || null,
      },
    });
  } catch (error) {
    return res.status(500).json({ error: 'Auth error' });
  }
});

    const token = tokenFromHeader || cookies.cb_token || '';
    const source = tokenFromHeader ? 'bearer' : (cookies.cb_token ? 'cookie' : 'none');

      console.log('[AUTH_ME_DEBUG]', { hasAuthHeader: Boolean(tokenFromHeader), hasCookie: Boolean(cookies.cb_token) });
    }
    if (!token) {
      res.set('Cache-Control', 'no-store, no-cache, must-revalidate');
      console.log('[AUTH_ME]', { user: null, source, status: 401 });

    }
